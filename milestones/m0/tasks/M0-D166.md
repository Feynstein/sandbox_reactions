### probe/mb.py (linux-pc, numpy; imports M0-D163's flash.py, tasks/M0-D163.md § Appendix)
```python
#!/usr/bin/env python3
"""M0-D166 mass-budget probe. Per dump (with its observers' snapshot beside it): the grid mass, the ledger's escaped, the
instantaneous outflow rate through each edge (sum over the edge row/column of Sigma * max(u_out, 0)), the unbound mass
(G-FIT's e = 1/2|u|^2 + eps_th + phi >= 0, and the Bernoulli 1/2|u|^2 + 2 eps_th + phi >= 0, gamma = 2) and, in radial shells
about the densest cell, the mass-weighted outward accelerations: gravity g_r = -dphi/dr, pressure p_r = -(1/Sigma) dPi/dr and
the radiation push f_r = kappa F_r / c_sb (LP flux from the dump's T, section 2.4.1), with the mean v_r. Then the side outflow
integrated over the dumps (trapezoid). Reads dumps, writes nothing.
  python3 -I mb.py <assets dir> <dump>..."""
import sys, json, math
import numpy as np
sys.path.insert(0, __file__.rsplit('/', 1)[0])
from flash import read_dump, state, PH

def phi_of(s, gas):
    H, W = s.shape
    src = np.where(gas, s, 0.0)
    yy = np.arange(-H + 1, H); xx = np.arange(-W + 1, W)
    Y, X = np.meshgrid(yy, xx, indexing='ij'); r = np.hypot(X, Y)
    with np.errstate(divide='ignore'):
        K = np.where(r > 0, -1.0 / r, -4 * math.log(1 + math.sqrt(2)))
    sh = (3 * H, 3 * W)
    out = np.fft.irfft2(np.fft.rfft2(src, sh) * np.fft.rfft2(K, sh), sh)
    return out[H - 1:2 * H - 1, W - 1:2 * W - 1]

def lam(R):
    return (2 + R) / (6 + 3 * R + R * R)

rows = []
for p in sys.argv[2:]:
    h, pl = read_dump(p); st = state(pl); s = st['s']; gas = s >= PH['sigma_vac']
    try:
        ob = json.load(open(p[:-4] + '.observers.json')); esc = float(ob['ledger']['last']['mass']['escaped'])
    except Exception:
        esc = float('nan')
    H, W = s.shape
    phi = phi_of(s, gas)
    u, v = st['u'], st['v']
    sm = np.where(gas, s, 0.0)
    out = dict(left=float((sm[:, 0] * np.maximum(-u[:, 0], 0)).sum()), right=float((sm[:, -1] * np.maximum(u[:, -1], 0)).sum()),
               bottom=float((sm[0, :] * np.maximum(-v[0, :], 0)).sum()), top=float((sm[-1, :] * np.maximum(v[-1, :], 0)).sum()))
    kin = 0.5 * (u * u + v * v)
    e = kin + st['eth'] + phi; b = kin + 2 * st['eth'] + phi
    iy, ix = np.unravel_index(np.argmax(sm), s.shape)
    yy, xx = np.mgrid[0:H, 0:W]; dx, dy = xx - ix, yy - iy; r = np.hypot(dx, dy)
    with np.errstate(invalid='ignore', divide='ignore'):
        ex, ey = np.where(r > 0, dx / r, 0), np.where(r > 0, dy / r, 0)
    vr = u * ex + v * ey
    gy, gx = np.gradient(phi); gr = -(gx * ex + gy * ey)
    Pi = st['pth'] + st['pc']; py, px = np.gradient(Pi)
    with np.errstate(invalid='ignore', divide='ignore'):
        pr = -(px * ex + py * ey) / s
    kap = PH['kappa0'] * (1 + np.maximum(st['X'][0], 0))
    E4 = PH['a_r'] * st['T'] ** 4; ey4, ex4 = np.gradient(E4)
    with np.errstate(invalid='ignore', divide='ignore'):
        R = np.hypot(ex4, ey4) / (kap * s * E4)
        fr = -lam(R) * (ex4 * ex + ey4 * ey) / s          # kappa F_r / c_sb per unit mass
    order = np.argsort(r[gas]); mcum = np.cumsum(sm[gas][order]); M = mcum[-1]
    r90 = float(r[gas][order][np.searchsorted(mcum, 0.9 * M)])
    shells = []
    for lo, hi in [(0.5, 1.0), (1.0, 1.5), (1.5, 2.5), (2.5, 99)]:
        m = gas & (r >= lo * r90) & (r < hi * r90) & np.isfinite(pr) & np.isfinite(fr)
        mm = sm[m].sum()
        if mm <= 0:
            shells.append(None); continue
        wavg = lambda a: float((sm[m] * a[m]).sum() / mm)
        shells.append(dict(m=float(mm), vr=wavg(vr), g=wavg(gr), p=wavg(pr), f=wavg(fr), ub=float(sm[m & (e >= 0)].sum())))
    rows.append(dict(step=h['step'], t=h['sim_time'], M=float(sm.sum()), esc=esc, out=out, ub_e=float(sm[e >= 0].sum()),
                     ub_b=float(sm[b >= 0].sum()), ub_out=float(sm[(e >= 0) & (vr > 0)].sum()), r90=r90, sh=shells,
                     L=float(ob['watch']['tracker']['objects'][0]['luminosity']) if esc == esc else float('nan')))
rows.sort(key=lambda x: x['t'])
cum = dict(left=0.0, right=0.0, bottom=0.0, top=0.0)
print('%7s %7s %7s %7s | %7s %7s %7s %7s (cum side) | %7s %7s %7s %5s %8s | shells [0.5,1) [1,1.5) [1.5,2.5) [2.5,): m vr | g p f (outward accel per mass)' % (
    'step', 't', 'M', 'esc', 'left', 'right', 'bottom', 'top', 'ub_e', 'ub_b', 'ub_out', 'R90', 'L'))
for i, r_ in enumerate(rows):
    if i:
        dt = r_['t'] - rows[i - 1]['t']
        for k in cum: cum[k] += 0.5 * dt * (r_['out'][k] + rows[i - 1]['out'][k])
    sh = ' '.join('[%5.0f %+5.2f | %+6.2f %+6.2f %+6.2f]' % (x['m'], x['vr'], x['g'], x['p'], x['f']) if x else '[  -  ]' for x in r_['sh'])
    print('%7d %7.2f %7.0f %7.1f | %7.1f %7.1f %7.1f %7.1f | %7.0f %7.0f %7.0f %5.0f %8.3g | %s' % (
        r_['step'], r_['t'], r_['M'], r_['esc'], cum['left'], cum['right'], cum['bottom'], cum['top'], r_['ub_e'], r_['ub_b'],
        r_['ub_out'], r_['r90'], r_['L'], sh))
```

#!/usr/bin/env python3
"""M0-D166: radial profile about the densest cell — Sigma, T, the radiation temperature T_rad = (L / (2 pi r c_sb a_r))^(1/4) from the
tracker's L, the isothermal a^2 = eps_th, the escape GM(<r)/r, v_r — mass-weighted per ring. python3 -I prof.py <assets> <dump>"""
import sys, json, math
import numpy as np
sys.path.insert(0, __file__.rsplit('/', 1)[0])
from flash import read_dump, state, PH
p = sys.argv[2]; h, pl = read_dump(p); st = state(pl); s = st['s']; gas = s >= PH['sigma_vac']; sm = np.where(gas, s, 0)
ob = json.load(open(p[:-4] + '.observers.json')); L = float(max(ob['watch']['tracker']['objects'], key=lambda o: float(o['mass']))['luminosity'])
iy, ix = np.unravel_index(np.argmax(sm), s.shape); yy, xx = np.mgrid[0:s.shape[0], 0:s.shape[1]]; r = np.hypot(xx - ix, yy - iy)
