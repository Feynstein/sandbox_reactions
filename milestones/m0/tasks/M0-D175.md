## Appendix — core.py (linux-pc scratch; imports M0-D163's flash.py, copied from build/m0-d167/)
```python
import sys, math
import numpy as np
sys.path.insert(0, __file__.rsplit('/', 1)[0])
from flash import read_dump, state, PH
for p in sys.argv[2:]:
    h, pl = read_dump(p); st = state(pl); s = st['s']; gas = s >= PH['sigma_vac']; sg = np.where(gas, s, 0.0); X = st['X']
    iy, ix = np.unravel_index(np.argmax(sg), s.shape)
    yy, xx = np.mgrid[0:s.shape[0], 0:s.shape[1]]; r = np.hypot(xx - ix, yy - iy)
    he = gas & (X[0] < 0.5); co = gas & (X[2:10].sum(0) > 0.5); si = gas & (X[6:10].sum(0) > 0.5); fe = gas & ((X[8] + X[9]) > 0.5)
    xc = X[:, iy, ix]; e = st['eps'][:, iy, ix]
    print(h['step'], h['sim_time'], st['T'][iy, ix], s[iy, ix], st['d'][iy, ix], st['eth'][iy, ix] / max(st['ec'][iy, ix], 1e-30),
          xc[0], xc[1], xc[2] + xc[3], xc[6] + xc[7], xc[8], e[0], e[1], e[4], e[5], st['enu'][iy, ix],
          sg[he].sum(), sg[co].sum(), sg[si].sum(), sg[fe].sum(), sg[r <= 2].sum(), sg[r <= 4].sum(), sg[r <= 8].sum(), sg.sum())
```
