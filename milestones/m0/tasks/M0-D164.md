## Appendix — the probe script (scratchpad copy, verbatim)

### probe/reach.py
```python
#!/usr/bin/env python3
"""M0-D164 (1): one step's heat reach. For a start dump and its one-step successors (full step; P5 alone), the outermost cell
(about the densest cell) whose T rose by more than 1 %, 10 %, x2 over the step, the cells heated past T_floor*2 from below it,
against light's reach c_sb*dt; X_H/X_Fe change. Usage: python3 -I reach.py <assets> <start.bin> <label>=<next.bin>..."""
import sys, json, numpy as np
sys.path.insert(0, __file__.rsplit('/', 1)[0])
from flash import read_dump, state, PH
h0, p0 = read_dump(sys.argv[2]); s0 = state(p0)
gas = s0['s'] >= PH['sigma_vac']
iy, ix = np.unravel_index(np.argmax(np.where(gas, s0['s'], 0)), gas.shape)
yy, xx = np.mgrid[0:gas.shape[0], 0:gas.shape[1]]; r = np.hypot(xx - ix, yy - iy)
for a in sys.argv[3:]:
    lab, p = a.split('=', 1); h1, p1 = read_dump(p); s1 = state(p1)
    dt = h1['sim_time'] - h0['sim_time']; light = PH['c_sb'] * dt
    g = gas & (s1['s'] >= PH['sigma_vac'])
    rat = np.where(g, s1['T'] / s0['T'], 1.0)
    out = []
    for f in (1.01, 1.1, 2.0):
        m = g & (rat > f); out.append(r[m].max() if m.any() else 0.0)
    m = g & (rat > 1.1); 
    dH = float(np.abs(s1['X'][0] - s0['X'][0])[g].max()); fe = float(s1['X'][8][g].max())
    print('%s step %d->%d dt %.5f c_sb*dt %.1f cells | outermost r with T x1.01 %.1f, x1.1 %.1f, x2 %.1f | max T ratio %.3g at r %.1f | heated mass (x1.1) %.3g | max |dX_H| %.2g, max X_Fe %.2g' % (
        lab, h0['step'], h1['step'], dt, light, *out, rat.max(), r[np.unravel_index(np.argmax(rat), rat.shape)], float(s1['s'][m].sum()), dH, fe))
```
