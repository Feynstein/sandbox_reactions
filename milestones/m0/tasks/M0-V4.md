### The probe module (`grav::v4probe`, 4 cases)
```rust
/// M0-V4's probes — appended to the scratch copy's grav.rs only, never the tree: P2 on non-square FFT sizes and the
/// whole default world against the twin and Newton's third law, and sub-Σ_vac cells as no source.
mod v4probe {
    use super::*;
    use sr_physics::registry::Physics;

    fn sigma_vac() -> f64 {
        Physics::shipped().unwrap().get("sigma_vac").unwrap()
    }

    /// |Σ m·g| / Σ m|g| over the box, m the masked source.
    fn ratio(c: &super::gpu::Cropped) -> f64 {
        let src = grav::source(&c.sigma, sigma_vac());
        let (mut nx, mut ny, mut scale) = (0.0f64, 0.0f64, 0.0f64);
        for ((&m, &x), &y) in src.iter().zip(&c.gx).zip(&c.gy) {
            nx += m * x;
            ny += m * y;
            scale += m * x.hypot(y);
        }
        nx.hypot(ny) / scale
    }

    /// Relative L1 of the GPU's φ, gx, gy against the twin's over the box.
    fn l1s(s: &super::gpu::Solved) -> (f64, f64, f64) {
        let c = s.cropped();
        let twin = Gravity::new(G).field(&c.sigma, sigma_vac(), s.area.w as usize, s.area.h as usize).unwrap();
        let l1 = |gpu: &[f64], twin: &[f64]| {
            gpu.iter().zip(twin).map(|(a, b)| (a - b).abs()).sum::<f64>() / twin.iter().map(|b| b.abs()).sum::<f64>()
        };
        (l1(&c.phi, &twin.phi), l1(&c.gx, &twin.gx), l1(&c.gy, &twin.gy))
    }

    fn run(what: &str, w: u32, h: u32, sigma: &[f64], limit: f64) {
        let gpu = crate::device();
        let solved = super::gpu::solve(&gpu, sigma, w, h, 1, true);
        let c = solved.cropped();
        let (p, x, y) = l1s(&solved);
        let b = &solved.area;
        println!(
            "SR-V4PROBE {what}: world {w}x{h}, box {}x{} at ({}, {}), FFT {:?}: rel L1 vs twin phi {p:.2e} gx {x:.2e} gy {y:.2e}; net-force ratio {:.3e}",
            b.w, b.h, b.x0, b.y0, b.fft, ratio(&c)
        );
        assert!(p <= limit && x <= limit && y <= limit, "{what}: GPU against the twin over {limit}");
    }

    #[test]
    fn wide_nonsquare_box_in_the_default_world() {
        let (w, h) = (600usize, 400usize);
        let mut sigma = vec![0.0; w * h];
        for j in 0..h {
            for i in 0..w {
                let r2 = ((i as f64 - 150.0).powi(2) + (j as f64 - 200.0).powi(2)) / 400.0;
                if r2 < 1.0 {
                    sigma[j * w + i] = (1.0 - r2).sqrt();
                }
                if (400..500).contains(&i) && (185..215).contains(&j) {
                    sigma[j * w + i] += 0.5;
                }
            }
        }
        run("wide 600x400", w as u32, h as u32, &sigma, 1e-3);
    }

    #[test]
    fn tall_nonsquare_box() {
        let (w, h) = (128usize, 512usize);
        let mut sigma = vec![0.0; w * h];
        for j in 100..420 {
            for i in 50..70 {
                sigma[j * w + i] = 0.3 + 0.001 * j as f64;
            }
        }
        run("tall 128x512", w as u32, h as u32, &sigma, 1e-3);
    }

    #[test]
    fn whole_default_world() {
        let (w, h) = (600usize, 400usize);
        let sigma: Vec<f64> = (0..w * h)
            .map(|k| {
                let (i, j) = ((k % w) as f64, (k / w) as f64);
                0.5 + 0.25 * (i / 37.0).sin() * (j / 23.0).cos()
            })
            .collect();
        run("whole world 600x400", w as u32, h as u32, &sigma, 1e-3);
    }

    #[test]
    fn sub_vacuum_cells_are_no_source() {
        // a faint disk (Σc = 1e-3, a = 20) in 0.9 Σ_vac everywhere else: the box is the disk's, and its vacuum cells,
        // masked, add nothing — unmasked they shift φ by ~4e-4 relative (my arithmetic), so a 1e-5 bound sees it.
        let (w, h) = (128usize, 128usize);
        let floor = 0.9 * sigma_vac();
        let mut sigma = vec![floor; w * h];
        for j in 0..h {
            for i in 0..w {
                let r2 = ((i as f64 - 64.0).powi(2) + (j as f64 - 64.0).powi(2)) / 400.0;
                if r2 < 1.0 {
                    sigma[j * w + i] = 1e-3 * (1.0 - r2).sqrt().max(0.01);
                }
            }
        }
        run("sub-vacuum background", w as u32, h as u32, &sigma, 1e-5);
    }
}
```
