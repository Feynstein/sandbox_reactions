# python3 -I v18_replay.py milestones/m0/logs/M0-V8.probe.log  (kept in the session scratchpad; the text is here)
import re, sys
rows, cur = [], None
for line in open(sys.argv[1]):
    m = re.match(r"V8PROBE frame step (\d+) t ([\d.e+-]+) dt \S+ box .*objects (\d+) L_box (\S+) LH_box (\S+)", line)
    if m:
        cur = dict(step=int(m[1]), t=float(m[2]), n=int(m[3]), Lbox=float(m[4])); rows.append(cur); continue
    m = re.match(r"V8PROBE   id \d+ M \S+ L (\S+) L_H (\S+) L_nuc (\S+) d_c (\S+) Munb/M (\S+) R \S+ v_r/v (\S+) "
                 r"T_c \S+ S_c \S+ X_H (\S+) 2P/W (\S+) raw Some\((\w+)\)", line)
    if m:
        cur.update(L=float(m[1]), LH=float(m[2]), Lnuc=max(0.0, float(m[3])), dc=float(m[4]), munb=float(m[5]),
                   vr=float(m[6]), XH=float(m[7]), pw=float(m[8]), raw=m[9])
CAMEL = {"failed_star": "FailedStar", "main_sequence": "MainSequence", "cloud": "Cloud", "white_dwarf": "WhiteDwarf",
         "collapsing": "Collapsing", "protostar": "Protostar", "planetary_nebula": "PlanetaryNebula", "h_shell": "HShell"}
def below(a, k, L): return L > 0 and a < k * L
def hydrogen_leads(LH, L, balance): return L > 0 and (0.5 * L <= LH <= 2 * L if balance else LH >= 0.5 * L)
def stage(r, L, ign, balance):
    if ign and r["dc"] >= 0.5 and below(r["Lnuc"], 0.01, L) and r["munb"] < 0.05: return "white_dwarf"
    if not ign and r["dc"] >= 0.5 and below(r["LH"], 0.01, L): return "failed_star"
    if ign and r["dc"] >= 0.5 and below(r["Lnuc"], 0.1, L) and r["munb"] >= 0.05: return "planetary_nebula"
    if ign and r["XH"] < 1e-3: return "h_shell"
    if hydrogen_leads(r["LH"], L, balance) and r["XH"] >= 1e-3: return "main_sequence"
    if not ign and r["vr"] <= -0.1: return "collapsing"
    if not ign and abs(r["vr"]) <= 0.05 and r["pw"] >= 0.8: return "protostar"
    return "cloud"
def run(name, light, balance, check=False):
    ign, ign_step, shown, streak, raw_seq, shown_seq, mismatches = False, None, None, (None, 0), [], [], 0
    for r in rows:
        L = light(r)
        if not ign and hydrogen_leads(r["LH"], L, balance): ign, ign_step = True, r["step"]
        s = stage(r, L, ign, balance)
        mismatches += check and CAMEL[s] != r["raw"]
        streak = (s, streak[1] + 1) if streak[0] == s else (s, 1)
        if streak[1] >= 3: shown = s
        if not raw_seq or raw_seq[-1][1] != s: raw_seq.append((r["step"], s))
        if not shown_seq or shown_seq[-1][1] != shown: shown_seq.append((r["step"], shown))
    tail = f" · raw stage against the engine's: {len(rows) - mismatches}/{len(rows)} frames agree" if check else ""
    print(f"{name}: ignition at step {ign_step}{tail}\n  raw:   {raw_seq}\n  shown: {shown_seq}")
run("today", lambda r: r["L"], False, check=True)
run("nearest object, ignition unchanged", lambda r: r["Lbox"], False)
run("nearest object, 0.5 L <= L_H <= 2 L", lambda r: r["Lbox"], True)
