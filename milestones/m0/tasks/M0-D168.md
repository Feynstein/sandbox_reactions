### cells.py

#!/usr/bin/env python3
"""M0-D168: the remnant's cells in a dump — Sigma, T, eps_th vs the floor, eps_cold, u, F, and the light leaving to vacuum.
Usage: python3 -I cells.py <assets dir> <dump> [<dump2>]   (needs flash.py's state() — copied in below, minus the reactions)"""
import json, struct, sys
import numpy as np
A = sys.argv[1]
PH = json.load(open(A + '/physics.json'))
EL = json.load(open(A + '/elements.json'))['species']
AZ = [(s['a'], s['z']) for s in EL]

def cold_u(x, k1, k2):
    y = k1 / k2 * np.sqrt(np.maximum(x, 0))
    g = np.where(y < 1e-3, y * y / 2 - y ** 3 / 3 + y ** 4 / 4, y - np.log1p(y))
    return x * 2 * k2 * k2 / k1 * g

def cold_p(x, k1, k2):
    with np.errstate(divide='ignore', invalid='ignore'):
        return np.where(x > 0, 1.0 / (1.0 / (k1 * x * x) + 1.0 / (k2 * x ** 1.5)), 0.0)

def read(path):
    b = open(path, 'rb').read(); n = struct.unpack('<I', b[8:12])[0]; h = json.loads(b[12:12 + n])
    w, ht = h['width'], h['height']; npl = (len(b) - 12 - n) // (4 * w * ht)
    pl = np.frombuffer(b, dtype='<f4', count=npl * w * ht, offset=12 + n).reshape(npl, ht, w)
    return h, pl

def state(pl):
    p = pl.astype(np.float64)
    s = np.maximum(p[0], 1e-30); mx, my, e = p[1], p[2], p[3]; X = p[4:14]
    mu = 1 / sum(X[i] * (z + 1) / a for i, (a, z) in enumerate(AZ)); ye = sum(X[i] * z / a for i, (a, z) in enumerate(AZ))
    ec = (cold_u(ye * s, PH['k1e'], PH['k2e']) + cold_u(X[9] * s, PH['k1n'], PH['k2n'])) / s
    kin = 0.5 * (mx * mx + my * my) / (s * s)
    eth_raw = e / s - kin - ec
    return dict(s=s, u=mx / s, v=my / s, e=e, ec=ec, kin=kin, eth_raw=eth_raw, mu=mu, floor=PH['t_floor'] / mu,
                T=mu * np.maximum(eth_raw, PH['t_floor'] / mu), pc=cold_p(ye * s, PH['k1e'], PH['k2e']))

h, pl = read(sys.argv[2]); st = state(pl)
print('planes', pl.shape[0], 'step', h['step'], 'dt', h['resume']['dt'])
s = st['s']; gas = s >= PH['sigma_vac']
print('gas cells', int(gas.sum()), 'mass', float(s[gas].sum()))
iy, ix = np.unravel_index(np.argmax(s), s.shape)
print('centre', ix, iy)
ys, xs = np.nonzero(gas)
print('gas bbox x', xs.min(), xs.max(), 'y', ys.min(), ys.max())
np.set_printoptions(linewidth=250, precision=4)
y0, y1, x0, x1 = max(ys.min() - 1, 0), ys.max() + 2, max(xs.min() - 1, 0), xs.max() + 2
if (y1 - y0) * (x1 - x0) <= 400:
    for name, a in [('Sigma', s), ('E', st['e']), ('eps_cold', st['ec']), ('kin', st['kin']), ('eps_th raw', st['eth_raw']),
                    ('floor T_f/mu', st['floor']), ('u', st['u']), ('v', st['v'])] + ([('F_x', pl[14].astype(float)), ('F_y', pl[15].astype(float)), ('rate', pl[16].astype(float))] if pl.shape[0] >= 17 else []):
        print('--', name); print(a[y0:y1, x0:x1])
for k in range(10):
    print('X', EL[k]['key'], pl[4 + k][iy, ix])
if len(sys.argv) > 3:
    h2, pl2 = read(sys.argv[3]); st2 = state(pl2)
    print('== diff to', h2['step'])
    for name, k in [('Sigma', 0), ('E', 3), ('mom_x', 1), ('mom_y', 2)]:

        if (y1 - y0) * (x1 - x0) <= 400: print(dd[y0:y1, x0:x1])

### hll.py (discarded: first-order HLL without MUSCL or the gravity source reads the centre losing 22 mass a step — far from the scheme's balance; kept for the record)
```python
#!/usr/bin/env python3
"""M0-D168: first-order HLL mass and energy fluxes (Davis speeds, §2.3.4) at every face of the remnant, from a dump's cells;
the centre cell's energy drain per step against the floor_added the ledger books. Estimate only (no MUSCL, no gravity source)."""
import sys, importlib.util
spec = importlib.util.spec_from_file_location('cells', sys.argv[3]); 
sys.argv = [sys.argv[0], sys.argv[1], sys.argv[2]]
import numpy as np, json, struct
exec(open(spec.origin).read().split("h, pl = read(sys.argv[2])")[0])
h, pl = read(sys.argv[2]); st = state(pl); s = st['s']; dt = h['resume']['dt']
eth = np.maximum(st['eth_raw'], st['floor']); P = s * eth + st['pc']; c = np.sqrt(2 * P / s)
E = pl[3].astype(float)
def hll(ax):
    # faces between i and i+1 along ax
    sl = [slice(None)] * 2; sr = [slice(None)] * 2; sl[ax] = slice(0, -1); sr[ax] = slice(1, None); sl, sr = tuple(sl), tuple(sr)
    un = st['u'] if ax == 1 else st['v']
    SL = np.minimum(un[sl] - c[sl], un[sr] - c[sr]); SR = np.maximum(un[sl] + c[sl], un[sr] + c[sr])
    def flux(q, fq):
        FL, FR = fq[sl], fq[sr]
        return np.where(SL >= 0, FL, np.where(SR <= 0, FR, (SR * FL - SL * FR + SL * SR * (q[sr] - q[sl])) / (SR - SL)))
    fm = flux(s, s * un); fe = flux(E, (E + P) * un)
    adv_m = np.where(SL >= 0, s[sl]*un[sl], np.where(SR <= 0, s[sr]*un[sr], (SR*s[sl]*un[sl] - SL*s[sr]*un[sr])/(SR-SL)))
    return fm, fe, adv_m
fmx, fex, amx = hll(1); fmy, fey, amy = hll(0)
iy, ix = np.unravel_index(np.argmax(s), s.shape)
# net into the centre: left face (ix-1|ix) flux +, right face (ix|ix+1) flux -, same in y
dm = fmx[iy, ix-1] - fmx[iy, ix] + fmy[iy-1, ix] - fmy[iy, ix]
de = fex[iy, ix-1] - fex[iy, ix] + fey[iy-1, ix] - fey[iy, ix]
print('centre', ix, iy, 'Sigma', s[iy, ix], 'E/Sigma', E[iy, ix]/s[iy, ix], 'eps_cold', st['ec'][iy, ix])
print('face mass flux (left,right,up,down):', fmx[iy, ix-1], fmx[iy, ix], fmy[iy-1, ix], fmy[iy, ix])
print('  its advective part:', amx[iy, ix-1], amx[iy, ix], amy[iy-1, ix], amy[iy, ix])
print('net into centre per step: mass %.4g  energy %.4g' % (dm * dt, de * dt))
print('energy the centre needs to keep E/Sigma = eps_cold+floor at its new Sigma: %.4g' % ((s[iy,ix] + dm*dt) * (st['ec'][iy,ix] + st['floor'][iy,ix]) - E[iy,ix] - de*dt))
# every gas cell: net energy and mass by first-order HLL per step
dM = np.zeros_like(s); dE = np.zeros_like(s)
dM[:, 1:] += fmx; dM[:, :-1] -= fmx; dM[1:, :] += fmy; dM[:-1, :] -= fmy
dE[:, 1:] += fex; dE[:, :-1] -= fex; dE[1:, :] += fey; dE[:-1, :] -= fey
gas = s >= 1e-6
y0, y1, x0, x1 = iy - 4, iy + 5, ix - 4, ix + 5
np.set_printoptions(linewidth=220, precision=3)
print('-- dM per step'); print(dM[y0:y1, x0:x1] * dt)
print('-- dE per step'); print(dE[y0:y1, x0:x1] * dt)
```

#!/usr/bin/env python3
"""M0-D168: replay §1.9.3's white_dwarf row over every observers snapshot under a run folder, today's (settled: |v_r| <= 0.05 v_dyn
and 2(P + K_t)/|W| >= 0.8) against the candidate (settled, or at rest with d_c >= D_MIN). Counts the frames whose raw stage the
candidate would turn to white_dwarf (raw not already black_hole / supernova / neutron_star / core_collapse / white_dwarf), and
prints the first and last such frame with its terms. No numpy. Usage: replay.py <m_ch> <d_min> <dir>..."""
import json, sys, glob, re, os
m_ch, d_min = float(sys.argv[1]), float(sys.argv[2])
EARLY = {'black_hole', 'supernova', 'neutron_star', 'core_collapse', 'white_dwarf'}
for root in sys.argv[3:]:
    for d in sorted(set(os.path.dirname(p) for p in glob.glob(root + '/**/s_*.observers.json', recursive=True))):
        snaps = sorted(glob.glob(d + '/s_*.observers.json'), key=lambda p: int(re.search(r's_(\d+)\.obs', p).group(1)))
        n = flips = 0; first = last = None; raws = {}
        for p in snaps:
            try: j = json.load(open(p))
            except Exception: continue
            stg = {s['id']: s for s in j['watch']['stages']}
            for o in j['watch']['tracker']['objects']:
                s = stg.get(o['id'])
                if not s: continue
                n += 1; f = lambda k: float(o.get(k, 0.0) or 0.0)
                raws[s['raw']] = raws.get(s['raw'], 0) + 1
                vd = f('radius') / f('tau_dyn') if f('tau_dyn') > 0 else 0.0
                rest = abs(f('v_r')) <= 0.05 * vd
                vir = 2 * (f('pressure') + f('kinetic_t')) / abs(f('potential')) if f('potential') else 0.0
                base = (s['ignited'] and f('d_c') >= 0.5 and f('l_nuc') < 0.01 * f('luminosity') and f('unbound') < 0.05 * f('mass')
                        and float(o['x_c'][9]) < 0.5 and f('mass') <= m_ch)
                today = base and rest and vir >= 0.8
                cand = base and rest and (vir >= 0.8 or f('d_c') >= d_min)
                if cand and not today and s['raw'] not in EARLY:
                    flips += 1
                    row = (j['step'], float(j['sim_time']), s['raw'], s['shown'], f('mass'), f('radius'), f('d_c'), vir, f('v_r') / vd if vd else 0,
                           f('l_nuc') / f('luminosity'), float(o['x_c'][0]), f('t_c'))
                    first = first or row; last = row
        tag = os.path.relpath(d, root)
        print('%-60s frames %6d flips %5d  raw %s' % (root + '/' + tag, n, flips, ' '.join('%s:%d' % kv for kv in sorted(raws.items()))))
        for lab, r in (('first', first), ('last', last)):
            if r: print('    %s: step %d t %.2f raw %s shown %s M %.0f R %.1f d_c %.3f 2(P+Kt)/|W| %.3f vr/vd %.3f Lnuc/L %.2g X_H,c %.4f T_c %.3g' % ((lab,) + r))
