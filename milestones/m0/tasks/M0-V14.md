```python
#!/usr/bin/env python3
"""M0-V14 probe: a dump's object — kinetic energy split radial/tangential about its centre, angular momentum, W, P from the snapshot."""
import json, sys
import numpy as np
sys.path.insert(0, sys.argv[1])
from dumpread import read_dump, phi_of, SIGMA_VAC

h, pl = read_dump(sys.argv[2])
snap = json.load(open(sys.argv[3]))
obj = snap['watch']['tracker']['objects'][0]
obj = {k: ([float(x) for x in v] if isinstance(v, list) and v and isinstance(v[0], str) else (float(v) if isinstance(v, str) else v)) for k, v in obj.items()}
s, mx, my = pl[0], pl[1], pl[2]
gas = s >= SIGMA_VAC
ht, w = s.shape
yy, xx = np.mgrid[0:ht, 0:w]
xc, yc = xx + 0.5, yy + 0.5
M = s[gas].sum()
px, py = mx[gas].sum(), my[gas].sum()
vcx, vcy = px / M, py / M
cmx, cmy = (s * xc)[gas].sum() / M, (s * yc)[gas].sum() / M
ux = np.where(gas, mx / np.where(gas, s, 1), 0.0); uy = np.where(gas, my / np.where(gas, s, 1), 0.0)
K = 0.5 * (s * (ux ** 2 + uy ** 2))[gas].sum()
phi = phi_of(s)
W = 0.5 * (s * phi)[gas].sum()
print(f"step {h.get('step')} t {h.get('sim_time')}: grid M {M:.3f} (snapshot object M {obj['mass']:.3f}), bulk v ({vcx:.4g}, {vcy:.4g}), centre of mass ({cmx:.2f}, {cmy:.2f}), snapshot centre {obj['centre']}")
print(f"K (dump, all gas) {K:.2f} · snapshot kinetic {obj['kinetic']:.2f} · W (dump, FFT, all gas) {W:.2f} · snapshot potential {obj['potential']:.2f} · snapshot pressure {obj['pressure']:.2f}")
for name, (cx, cy) in (('snapshot centre (max-Sigma cell)', obj['centre']), ('centre of mass', (cmx, cmy))):
    rx, ry = xc - cx, yc - cy
    r = np.hypot(rx, ry); r = np.where(r > 0, r, 1.0)
    vx, vy = ux - vcx, uy - vcy
    vr = (vx * rx + vy * ry) / r
    vt = (vy * rx - vx * ry) / r
    Kr = 0.5 * (s * vr ** 2)[gas].sum(); Kt = 0.5 * (s * vt ** 2)[gas].sum()
    Lz = (s * (rx * vy - ry * vx))[gas].sum()
    vr_mw = (s * vr)[gas].sum() / M
    print(f"about the {name} ({cx:.2f}, {cy:.2f}): K_radial {Kr:.2f}, K_tangential {Kt:.2f} (share {Kt / (Kr + Kt):.3f}), L_z {Lz:.2f}, mass-weighted v_r {vr_mw:.4g}")
    print("   r band   mass    <v_t>(mass-w)  v_circ=sqrt(M(<r)/r)  <v_r>(mass-w)  Sigma_mean")
    edges = [0, 2, 5, 10, 20, 40, 60, 85, 120, 200, 400]
    for a, b in zip(edges[:-1], edges[1:]):
        band = gas & (r >= a) & (r < b)
        if not band.any(): continue
        m = s[band].sum(); inner = s[gas & (r < b)].sum()
        print(f"  [{a:3d},{b:3d})  {m:8.2f}  {(s * vt)[band].sum() / m:10.4f}  {np.sqrt(inner / b):10.4f}  {(s * vr)[band].sum() / m:10.4f}  {s[band].mean():.4g}")
```

```bash
cal_item() {  # step, label, then calibrate's arguments; --resume goes to the step START names

  todo "$n" || return 0
  stamp "$n calibrate $label"
```
