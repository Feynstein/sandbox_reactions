cal() { build/target/release/sandbox-reactions calibrate --adapter "RTX 4090" --out assets/calibration.json --checkpoint-every 3000 "$@"; }
cal_item() {  # step, label, then calibrate's arguments; --resume goes to the step START names
  local n=$1 label=$2; shift 2
  todo "$n" || return 0
  stamp "$n calibrate $label"
  if [ -n "$RESUME" ] && [ "$n" -eq "$START" ]; then cal "$@" --resume "$RESUME"; else cal "$@"; fi
}
same_dumps() {  # the re-runs against the recorded runs, dump by dump

    done
  done
}
