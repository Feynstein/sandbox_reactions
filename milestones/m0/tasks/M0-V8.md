## The dump reader (f64, contract §2.3's EOS, independent of the engine)
`python3 -I dump_dc.py <s_*.bin>…` (`K1E=5.325` for P2's dumps): the densest cell, first in row-major order — Σ, T (floored at
T_floor as P8 does), X_H, d_c = (Π_e + Π_n)/(Π_th + Π_e + Π_n), Π_th, Π_e, the grid's mass. μ and Y_e from elements.json
(1/μ = Σ X(Z+1)/A, Y_e = Σ X·Z/A), the cold energy in eos.rs's closed form.
```python
# M0-V8 probe (read-only): the max-Σ cell of each dump, read in f64 with contract §2.3's EOS, independent of the engine.
# usage: python3 -I dump_dc.py <dump.bin>...   prints step, sim_time, Σ_c, T_c, X_H,c, d_c = Π_cold/Π, Π_th, Π_e, box mass
import json, math, struct, sys

import os
K1E, K2E, K1N, K2N, T_FLOOR = float(os.environ.get("K1E", "21.3")), 134.0, 2.67, 58.2, 0.05
A = [1, 4, 12, 16, 20, 24, 28, 32, 56, 1]
Z = [1, 2, 6, 8, 10, 12, 14, 16, 26, 0]


def cold_p(x, k1, k2):
    if x <= 0 or k1 <= 0 or k2 <= 0:
        return 0.0
    return 1.0 / (1.0 / (k1 * x * x) + 1.0 / (k2 * x ** 1.5))


def cold_u(x, k1, k2):  # u(x) = x ∫0^x P(s)/s² ds, closed form (eos.rs cold_energy_exact)
    if x <= 0 or k1 <= 0 or k2 <= 0:
        return 0.0
    y = k1 / k2 * math.sqrt(x)
    g = sum((1 if k % 2 == 0 else -1) * y ** k / k for k in range(2, 10)) if y < 1e-3 else y - math.log1p(y)
    return x * 2.0 * k2 * k2 / k1 * g


def read(path):
    raw = open(path, "rb").read()
    assert raw[:8] == b"SRDUMP01", path
    n = struct.unpack("<I", raw[8:12])[0]
    head = json.loads(raw[12:12 + n].decode("utf-8"))
    w, h = head["width"], head["height"]
    cells = w * h
    off = 12 + n
    planes = []
    for c in range(14):
        planes.append(struct.unpack_from("<%df" % cells, raw, off + 4 * cells * c))
    return head, w, h, planes


for path in sys.argv[1:]:
    head, w, h, p = read(path)
    sig = p[0]
    best = max(range(w * h), key=lambda i: (sig[i], -i))  # the largest Σ, first in row-major order
    s = sig[best]
    x = [p[4 + k][best] for k in range(10)]
    inv_mu = sum(x[k] * (Z[k] + 1) / A[k] for k in range(10))
    y_e = sum(x[k] * Z[k] / A[k] for k in range(10))
    u = (p[1][best] / s, p[2][best] / s)
    eps_cold = (cold_u(y_e * s, K1E, K2E) + cold_u(x[9] * s, K1N, K2N)) / s
    eps_th = p[3][best] / s - 0.5 * (u[0] ** 2 + u[1] ** 2) - eps_cold
    t = max(eps_th, T_FLOOR * inv_mu) / inv_mu
    pi_th = s * eps_th
    pi_e = cold_p(y_e * s, K1E, K2E)
    pi_n = cold_p(x[9] * s, K1N, K2N)
    d = (pi_e + pi_n) / (pi_th + pi_e + pi_n)
    mass = sum(sig)
    print(f"step {head['step']:>6} t {head['sim_time']:8.3f}  Σc {s:8.4f}  Tc {t:8.3f}  X_H,c {x[0]:.4f}  d_c {d:.4f}"
          f"  Π_th {pi_th:9.3f}  Π_e {pi_e:9.3f}  cell ({best % w},{best // w})  mass {mass:.3f}")
```
