### rows.py
```python
#!/usr/bin/env python3
"""M0-D167: per observers snapshot of the heaviest object, in a t window: shown stage, M, R90, M_unb/M, env, v_r, L, L_nuc, L_N,
T_c, Sigma_c, d_c, X_c (H He CO SiS Fe n), escaped, floor_added. Usage: rows.py <run dir> [t0 t1 [every]]"""
import json, sys, glob, re
d = sys.argv[1]; t0 = float(sys.argv[2]) if len(sys.argv) > 2 else -1; t1 = float(sys.argv[3]) if len(sys.argv) > 3 else 1e9
ev = int(sys.argv[4]) if len(sys.argv) > 4 else 1
snaps = sorted(glob.glob(d + '/dumps/s_*.observers.json'), key=lambda p: int(re.search(r's_(\d+)', p).group(1)))
print('%6s %7s %-13s %5s %6s %5s %5s %6s %8s %8s %9s %6s %6s %4s | %5s %5s %5s %5s %6s %5s | %6s %9s' % ('step','t','stage','nobj','M','R90','unb','v_r','L','L_nuc','L_N','T_c','Sig_c','d_c','X_H','X_He','X_CO','X_SiS','X_Fe','X_n','esc','floor'))
k = 0
for p in snaps:
    try: j = json.load(open(p))
    except Exception: continue
    t = float(j['sim_time'])
    if t < t0 or t > t1: continue
    k += 1
    if (k - 1) % ev: continue
    objs = j['watch']['tracker']['objects']
    if not objs: print('%6d %7.2f (no object)' % (j['step'], t)); continue
    o = max(objs, key=lambda x: float(x['mass'])); st = {s['id']: s['shown'] for s in j['watch']['stages']}.get(o['id'], '?')
    f = lambda key: float(o[key]); x = [float(v) for v in o['x_c']]; lg = [float(v) for v in o['l_group']]
    M = f('mass')
    print('%6d %7.2f %-13s %5d %6.0f %5.1f %5.3f %6.2f %8.3g %8.3g %9.3g %6.1f %6.1f %4.2f | %5.3f %5.3f %5.3f %5.3f %6.4f %5.3f | %6.0f %9.4g' % (
        j['step'], t, st, len(objs), M, f('radius'), f('unbound') / M, f('v_r'), f('luminosity'), f('l_nuc'), lg[6], f('t_c'), f('sigma_c'), f('d_c'),
        x[0], x[1], x[2] + x[3], x[6] + x[7], x[8], x[9], float(j['ledger']['last']['mass']['escaped']), float(j['ledger']['last']['energy']['floor_added'])))
```

### probe.py
```python
#!/usr/bin/env python3
"""M0-D167 event probe: per dump, about the densest cell c — c's identity and composition, eps_th, eps_cold, eps_th/eps_cold,
the hold's mask (count, mass), mass inside r (2, 4, 8, 16, 32, 64), hydrogen mass inside r <= 4 and the inward H mass flux
through the r = 4 ring, the cores by composition (Fe+n > 0.5, Si+ > 0.5, C/O+ > 0.5, X_n > 0.5) against the ceiling, and the
star's unbound mass (1/2|u|^2 + eps_th + phi > 0 not computed: phi not in the dump). python3 -I probe.py <assets> <dump>..."""
import sys, math, json, struct
import numpy as np
sys.path.insert(0, __file__.rsplit('/', 1)[0])
from flash import read_dump, state, PH
def mask_of(path, h):
    if not h.get('resume', {}).get('held'): return None
    b = open(path, 'rb').read(); n = struct.unpack('<I', b[8:12])[0]; w, ht = h['width'], h['height']
    off = 12 + n + 17 * w * ht * 4
    return np.frombuffer(b, dtype='<f4', count=w * ht, offset=off).reshape(ht, w)
for p in sys.argv[2:]:
    h, pl = read_dump(p); st = state(pl); s = st['s']; gas = s >= PH['sigma_vac']; sg = np.where(gas, s, 0.0); X = st['X']
    iy, ix = np.unravel_index(np.argmax(sg), s.shape)
    yy, xx = np.mgrid[0:s.shape[0], 0:s.shape[1]]; r = np.hypot(xx - ix, yy - iy)
    with np.errstate(invalid='ignore', divide='ignore'):
        vr = np.where(r > 0, ((xx - ix) * st['u'] + (yy - iy) * st['v']) / r, 0.0)
    Mr = [float(sg[r <= R].sum()) for R in (2, 4, 8, 16, 32, 64)]
    MH4 = float((sg * X[0])[r <= 4].sum())
    ring = gas & (r > 3.5) & (r <= 4.5)
    fluxH = float(-(sg * X[0] * vr)[ring].sum() / 1.0)  # inward H mass per t.u. through ~the r=4 ring (unit width)
    fluxM = float(-(sg * vr)[ring].sum())
    fe = gas & ((X[8] + X[9]) > 0.5); si = gas & (X[6:10].sum(0) > 0.5); co = gas & (X[2:10].sum(0) > 0.5); nn = gas & (X[9] > 0.5)
    m = mask_of(p, h); nh = int((m > 0.5).sum()) if m is not None else 0; mh = float(sg[m > 0.5].sum()) if m is not None else 0.0
    ec = st['ec'][iy, ix]; et = st['eth'][iy, ix]
    xc = X[:, iy, ix]
```

```python
#!/usr/bin/env python3
"""M0-D167: the burn clock in a dump, by radius about c: per ring, mass, H mass, the raw nuclear power per group (sum Sigma eps_g),
the capped power sum Sigma min(qdot, q_max) split as the raw shares, q_max = eps_th c_s / K (c_s = sqrt(2 eps_th)), the clock's
factor (raw/capped), and t_H = X_H q_H / (qdot_H capped) — the time the clock gives a cell to burn its hydrogen. python3 -I clock.py <assets> <dump>..."""
import sys, math
import numpy as np
sys.path.insert(0, __file__.rsplit('/', 1)[0])
from flash import read_dump, state, PH
K = PH['burn_k']; G = ['H', 'He', 'C', 'Ne', 'O', 'Si', 'N']
for p in sys.argv[2:]:
    h, pl = read_dump(p); st = state(pl); s = st['s']; gas = s >= PH['sigma_vac']; sg = np.where(gas, s, 0.0); X = st['X']
    iy, ix = np.unravel_index(np.argmax(sg), s.shape)
    yy, xx = np.mgrid[0:s.shape[0], 0:s.shape[1]]; r = np.hypot(xx - ix, yy - iy)
    eps = st['eps']; qd = np.clip(eps[:6], 0, None).sum(0); qmax = st['eth'] * np.sqrt(2 * st['eth']) / K
    cap = np.where(qd > qmax, qmax / np.maximum(qd, 1e-300), 1.0)
    with np.errstate(invalid='ignore', divide='ignore'):
        vr = np.where(r > 0, ((xx - ix) * st['u'] + (yy - iy) * st['v']) / r, 0.0)
        tH = np.where(eps[0] > 0, X[0] * PH['q_h'] / (eps[0] * cap), np.inf)
        tin = np.where(vr < 0, r / -vr, np.inf)
    print('%s step %d t %.2f c (%d,%d)' % (p.rsplit('/', 1)[-1], h['step'], h['sim_time'], ix, iy))
    print('  %-7s %7s %7s %6s | %9s %9s %9s %9s %9s | %9s %7s | %6s %8s %8s %6s' % ('ring', 'M', 'M_H', 'T', 'rawL_H', 'rawL_He', 'rawL_O', 'rawL_Si', 'rawL_all', 'capL_all', 'capL_H', 'medfac', 't_H(med)', 't_in', 'vr'))
    for a, b in [(0, 1.5), (1.5, 3), (3, 4.5), (4.5, 6), (6, 8), (8, 12), (12, 16), (16, 24), (24, 32), (32, 64), (64, 400)]:
        m = gas & (r >= a) & (r < b)
        if not m.any(): continue
        rawg = [(s * eps[g])[m].sum() for g in range(6)]
        capall = (s * qd * cap)[m].sum(); capH = (s * eps[0] * cap)[m].sum()
        fac = np.median(np.where(cap[m] > 0, 1 / cap[m], np.inf))
        hm = m & (X[0] > 0.01)
        tHm = float(np.median(tH[hm])) if hm.any() else float('nan')
        vrm = float((s * vr)[m].sum() / s[m].sum())
        tinm = (0.5 * (a + b)) / -vrm if vrm < 0 else float('inf')
        print('  %4.1f-%-3.0f %7.0f %7.0f %6.1f | %9.3g %9.3g %9.3g %9.3g %9.3g | %9.3g %7.3g | %6.3g %8.3g %8.3g %6.2f' % (a, b, s[m].sum(), (s * X[0])[m].sum(),
              float((s * st['T'])[m].sum() / s[m].sum()), rawg[0], rawg[1], rawg[4], rawg[5], sum(rawg), capall, capH, fac, tHm, tinm, vrm))
```
