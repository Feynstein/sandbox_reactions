```js
for (const [label, extra] of [['as smoke.mjs launches it', {}], ['with chromiumSandbox: true', { chromiumSandbox: true }]]) {
  const browser = await chromium.launch({ executablePath: chrome, headless: true, args: chromeArgs, ...extra });
  try {
    const page = await browser.newPage();
    await page.goto('chrome://version');
    const cmd = await page.evaluate(() => document.querySelector('#command_line')?.textContent ?? '(no #command_line)');

    console.log(`${label}: Chrome ${browser.version()} · --no-sandbox ${sandboxOff ? 'PRESENT' : 'absent'}`);
    console.log(`  command line: ${cmd}`);
  } finally {
    await browser.close();
  }
}
```
