```python
#!/usr/bin/env python3
"""M0-D163 flash probe: per dump, the state recomputed from the planes with the run's assets (§2.3, §1.6.1, §2.5.3) —
T, d = Pi_cold/Pi, each group's specific heating eps_g = sum q_r omega_r and eps_nu, at the centre (the densest cell), at the
hottest cell, and summed over the world (L_g = sum Sigma eps_g). Reads dumps, writes nothing. Run with -I from the repo root:
  python3 -I flash.py <assets dir> <dump> [<dump> ...]"""
import json, struct, sys, math
import numpy as np

A = sys.argv[1]
PH = json.load(open(A + '/physics.json'))
RX = json.load(open(A + '/reactions.json'))['records']
EL = json.load(open(A + '/elements.json'))['species']
KEYS = [s['key'] for s in EL]
AZ = [(s['a'], s['z']) for s in EL]
GROUPS = ['H', 'He', 'C', 'Ne', 'O', 'Si', 'N']

def val(v):
    return PH[v] if isinstance(v, str) else v

RECS = []
for r in RX:
    if not r['enabled']:
        continue
    rt = r['rate']
    terms = [(t[1] * PH[t[0]], val(t[2]), t[3]) for t in rt['terms']]
    t_thr = rt['t_thr_factor'] * terms[0][0] if terms else None
    RECS.append(dict(id=r['id'], g=GROUPS.index(r['group']), q=r['q_ratio'] * PH['q_h'], a=rt['a'],
                     inp=[(KEYS.index(k), n) for (k, _), n in zip(r['inputs'], rt['orders'])], terms=terms, t_thr=t_thr,
                     coef=val(rt['a_coef']), gate=rt['gate']))

def f_of(rec, T):
    if rec['t_thr'] is None:
        return np.ones_like(T)
    out = np.zeros_like(T)
    for tk, nu, w in rec['terms']:
        u = np.log(np.maximum(T, 1e-30) / tk)
        out += np.exp(np.log(w) - 2.0 / 3.0 * u - (3 * nu + 2) * np.expm1(-u / 3.0))
    return np.where(T >= rec['t_thr'], out, 0.0)

def cold_p(x, k1, k2):
    with np.errstate(divide='ignore', invalid='ignore'):
        return np.where(x > 0, 1.0 / (1.0 / (k1 * x * x) + 1.0 / (k2 * x ** 1.5)), 0.0)

def cold_u(x, k1, k2):
    y = k1 / k2 * np.sqrt(np.maximum(x, 0))
    g = np.where(y < 1e-3, y * y / 2 - y ** 3 / 3 + y ** 4 / 4, y - np.log1p(y))
    return x * 2 * k2 * k2 / k1 * g

def read_dump(path):
    b = open(path, 'rb').read()
    assert b[:8] == b'SRDUMP01'
    n = struct.unpack('<I', b[8:12])[0]
    h = json.loads(b[12:12 + n])
    w, ht = h['width'], h['height']
    pl = np.frombuffer(b, dtype='<f4', count=14 * w * ht, offset=12 + n).reshape(14, ht, w).astype(np.float64)
    return h, pl

def state(pl):
    s = np.maximum(pl[0], 1e-30); mx, my, e = pl[1], pl[2], pl[3]; X = pl[4:14]
    inv_mu = sum(X[i] * (z + 1) / a for i, (a, z) in enumerate(AZ)); mu = 1 / inv_mu
    ye = sum(X[i] * z / a for i, (a, z) in enumerate(AZ)); xn = X[9]
    ec = (cold_u(ye * s, PH['k1e'], PH['k2e']) + cold_u(xn * s, PH['k1n'], PH['k2n'])) / s
    kin = 0.5 * (mx * mx + my * my) / (s * s)
    eth = np.maximum(e / s - kin - ec, PH['t_floor'] / mu)
    T = mu * eth
    pth = s * eth; pc = cold_p(ye * s, PH['k1e'], PH['k2e']) + cold_p(xn * s, PH['k1n'], PH['k2n'])
    eps = np.zeros((len(GROUPS),) + s.shape)
    for r in RECS:
        om = r['coef'] * s ** r['a'] * f_of(r, T)
        for i, n in r['inp']:
            om = om * np.maximum(X[i], 0) ** n
        if r['gate']:
            sg = PH[r['gate']['sigma_key']]
            om = om * np.where(ye * s > sg, (ye * s / sg - 1) ** r['gate']['p'], 0.0)
        eps[r['g']] += r['q'] * om
    enu = np.where(T >= PH['t_nu'], PH['a_nu'] * (T / PH['t_nu']) ** PH['m_nu'], 0.0)
    return dict(s=s, u=mx / s, v=my / s, X=X, mu=mu, ye=ye, eth=eth, ec=ec, T=T, pth=pth, pc=pc, d=pc / (pth + pc), eps=eps, enu=enu)

def row(path, prev_t):
    h, pl = read_dump(path)
    st = state(pl)
    s, T = st['s'], st['T']
    gas = s >= PH['sigma_vac']
    iy, ix = np.unravel_index(np.argmax(np.where(gas, s, 0)), s.shape)
    ty, tx = np.unravel_index(np.argmax(np.where(gas, T, 0)), s.shape)
    yy, xx = np.mgrid[0:s.shape[0], 0:s.shape[1]]
    r = np.hypot(xx - ix, yy - iy)
    core = gas & (r <= 4.5)
    with np.errstate(invalid='ignore', divide='ignore'):
        vr = np.where(r > 0, ((xx - ix) * st['u'] + (yy - iy) * st['v']) / r, 0.0)
    vr_core = float((s * vr)[core].sum() / s[core].sum())
    L = (s * st['eps']).sum(axis=(1, 2)); Lnu = float((s * st['enu']).sum())
    epsc = st['eps'][:, iy, ix]; ec_nuc = float(epsc.sum())
    X = st['X'][:, iy, ix]
    dt = h['resume'].get('dt') if 'resume' in h else None
    # heating time at c: eps_th over the net specific power (inf when cooling)
    net = ec_nuc - float(st['enu'][iy, ix])
    t_heat = float(st['eth'][iy, ix] / net) if net > 0 else float('inf')
    # sound-crossing time of a 4.5-cell core at c: R / c, c^2 = 2 Pi / Sigma
    c_s = math.sqrt(2 * (st['pth'][iy, ix] + st['pc'][iy, ix]) / s[iy, ix])
    return dict(step=h['step'], t=h['sim_time'], dt=dt, c=(int(ix), int(iy)), sig_c=float(s[iy, ix]), T_c=float(T[iy, ix]),
                d_c=float(st['d'][iy, ix]), mu_c=float(st['mu'][iy, ix]), X_c=[float(v) for v in X], eps_c=[float(v) for v in epsc],
                enu_c=float(st['enu'][iy, ix]), t_heat=t_heat, t_sc=4.5 / c_s, vr_core=vr_core,
                hot=(int(tx), int(ty)), T_max=float(T[ty, tx]), sig_hot=float(s[ty, tx]), d_hot=float(st['d'][ty, tx]),
                X_hot=[float(v) for v in st['X'][:, ty, tx]], eps_hot=float(st['eps'][:, ty, tx].sum()),
                L=[float(v) for v in L], Lnu=Lnu, m_fe=float(s[gas & (st['X'][8] > 0.5)].sum()),
                m_hot=float(s[gas & (T > PH['t_h'])].sum()), m_core=float(s[core].sum()))

if __name__ == '__main__':
    fmt = sys.argv[2] if sys.argv[2] in ('json', 'table') else 'table'
    paths = sys.argv[3:] if sys.argv[2] in ('json', 'table') else sys.argv[2:]
    for p in paths:
        r = row(p, None)
        if fmt == 'json':
            print(json.dumps(r)); continue
        X = r['X_c']
        print('%3d t %.4f dt %.5f | c %s sig %7.3f T %7.2f d %.3f | X H %.3g He %.3g C %.3g O %.3g Ne %.3g Mg %.3g Si %.3g S %.3g Fe %.3g | '
              'eps_c H %.3g He %.3g C %.3g Ne %.3g O %.3g Si %.3g nu %.3g | t_heat %.3g t_sc %.3g vr %.3f | hot %s T %.1f sig %.2f d %.2f eps %.3g | '
              'L H %.3g He %.3g C %.3g Ne %.3g O %.3g Si %.3g nu %.3g | m_hot %.1f m_fe %.1f m_core %.1f' % (
            r['step'], r['t'], r['dt'] or 0, r['c'], r['sig_c'], r['T_c'], r['d_c'], X[0], X[1], X[2], X[3], X[4], X[5], X[6], X[7], X[8],
            *r['eps_c'][:6], r['enu_c'], r['t_heat'], r['t_sc'], r['vr_core'], r['hot'], r['T_max'], r['sig_hot'], r['d_hot'], r['eps_hot'],
            *r['L'][:6], r['Lnu'], r['m_hot'], r['m_fe'], r['m_core']))
```

### probe/mkmap.py
```python
#!/usr/bin/env python3
"""M0-D163: disk scenes equal to the preset cloud at mass M (a = 40 sqrt(M/M0) capped 150, Sigma_c 3M/(2 pi a^2), the Sun's mix, at rest)
at the rule's U_th = min(0.5, 0.1 max(1,M/M0)^2)|W|, W from one FFT solve of the centre-sampled cells (§1.4.2's kernel, G = 1) scaled to M.
Writes map_<M>.json into the cwd; with --check prints the T0 it computes for the 5,902.206 preset at 0.1 (the real one: 6.28415)."""
import json, math, sys
import numpy as np
M0 = 2 * math.pi * 1600 / 3
MIX = {"H": 0.7346, "He": 0.2485, "O": 0.0077, "C": 0.0092}
AZ = {"H": (1, 1), "He": (4, 2), "C": (12, 6), "O": (16, 8)}
inv_mu = sum(x * (AZ[k][1] + 1) / AZ[k][0] for k, x in MIX.items()); mu = 1 / inv_mu
def t0(M, ratio):
    a = min(40 * math.sqrt(M / M0), 150.0); sc = 3 * M / (2 * math.pi * a * a)
    n = int(2 * a) + 3; w = h = n
    yy, xx = np.mgrid[0:h, 0:w]; r2 = (xx + .5 - w / 2) ** 2 + (yy + .5 - h / 2) ** 2
    s = np.where(r2 < a * a, sc * np.sqrt(np.maximum(1 - r2 / (a * a), 0)), 0.0); s *= M / s.sum()
    ky, kx = np.meshgrid(np.fft.fftfreq(2 * h, 1 / (2 * h)), np.fft.fftfreq(2 * w, 1 / (2 * w)), indexing='ij')
    with np.errstate(divide='ignore'): k = -1.0 / np.hypot(kx, ky)
    k[0, 0] = -3.525494348078172
    pad = np.zeros((2 * h, 2 * w)); pad[:h, :w] = s
    phi = np.fft.irfft2(np.fft.rfft2(pad) * np.fft.rfft2(k), s=pad.shape)[:h, :w]
    W = 0.5 * (s * phi).sum()
    return mu * ratio * abs(W) / M, a, sc
if sys.argv[1] == '--check':
    t, a, sc = t0(5902.206362570599, 0.1); print('check: T0 %.5f (real preset 6.28415), a %.3f' % (t, a)); sys.exit()
for M in map(float, sys.argv[1:]):
    ratio = min(0.5, 0.1 * max(1.0, M / M0) ** 2); t, a, sc = t0(M, ratio)
    d = {"version": 1, "world": {"width": 600, "height": 400, "edge": "leave"}, "slowdown": True,
         "switches": {"gravity": True, "hydro": True, "heat": True, "reactions": True, "neutrinos": True, "sinks": True},
         "objects": [{"kind": "disk", "x": 300, "y": 200, "a": a, "sigma_c": sc, "t": t, "composition": MIX, "v": [0, 0]}]}
    json.dump(d, open('map_%d.json' % M, 'w')); print('M %d ratio %.3f a %.2f T0 %.4f' % (M, ratio, a, t))
```
