## probe/radial.py (beside lives.md's probe/dumpread.py)
```python
#!/usr/bin/env python3
"""M0-D158: per dump — gas + sinks, mass by r from the centre (300, 200) in 50-cell bins, and the 16-cell band's bound gas
(nec: 1/2|u|^2+phi<0) mass, its densest cell and the nearest bound gas at Sigma >= 1e-2. Reads dumps, writes nothing."""
import json, sys, glob, re
import numpy as np
sys.path.insert(0, __file__.rsplit('/', 1)[0])
from dumpread import read_dump, phi_of, edge_distance, SIGMA_VAC, C_SB
def one(path):
    h, pl = read_dump(path); s, mx, my = pl[0], pl[1], pl[2]
    with np.errstate(divide='ignore', invalid='ignore'):
        u2 = np.where(s > 0, (mx*mx + my*my)/(s*s), 0.0)
    sinks = [dict(x=k['x'], y=k['y'], m=k['m'], r_acc=max(2.0, 2*k['m']/C_SB**2)) for k in h.get('sinks', [])]
    phi = phi_of(s, sinks); d = edge_distance(*s.shape); gas = s >= SIGMA_VAC
    yy, xx = np.mgrid[0:s.shape[0], 0:s.shape[1]]; r = np.hypot(xx + .5 - 300, yy + .5 - 200)
    bins = [float(s[gas & (r >= a) & (r < a + 50)].sum()) for a in (0, 50, 100, 150, 200)] + [float(s[gas & (r >= 250)].sum())]
    b = gas & (0.5*u2 + phi < 0); bb = b & (d < 16)
    f2 = b & (s >= 1e-2)
    return dict(step=h['step'], t=round(h['sim_time'], 2), gas=round(float(s[gas].sum()), 0), sink=round(sum(k['m'] for k in h.get('sinks', [])), 0),
                r50=[round(x) for x in bins], band_m=round(float(s[bb].sum()), 1), band_smax=round(float(s[bb].max()), 4) if bb.any() else 0,
                d_1e2=float(d[f2].min()) if f2.any() else None)
if __name__ == '__main__':
    for dd in sys.argv[1:]:
        for f in sorted(glob.glob(dd + '/s_*.bin'), key=lambda p: int(re.search(r's_(\d+)\.bin', p).group(1))):
            print(dd, json.dumps(one(f)), flush=True)
```
